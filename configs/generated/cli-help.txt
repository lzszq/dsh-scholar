Options (runner-profile):
  --kernel <string> — Kernel endpoint the runner claims jobs from. (default: http://127.0.0.1:7412)
  --mode <string> — Execution mode; formal kinds are container-only (runtime enforced, execution-runtime.md §1). (default: subprocess)
  --poll-ms <number> — Poll interval between claim cycles.
  --timeout-ms <number> — Per-job execution timeout.
  --heartbeat-ms <number> — Lease heartbeat interval while a job executes.
  --cancel-poll-ms <number> — Cancel-watcher poll interval (terminates the real execution).
  --owner <string> — Lease owner identity; a random runner-<id> is generated when empty.
  --key-file <string> — Ed25519 signing key file (0600); an ephemeral key is generated when absent.
  --fleet-server <number> — Fleet server listen port (0 = ephemeral); when provided the runner runs RemoteFleetServer instead of the local claim loop.
  --agent <string> — Fleet server base URL; when provided the runner runs as a remote fleet agent instead of the local claim loop.
  --agent-id <string> — Agent identity registered with the fleet server; generated agent-<id> when empty.
  --target-id <string> — Opaque target id used for exact-match dispatch. Local mode defaults by mode to target_local_process_v1 or target_local_docker_v1; fleet agents must advertise their configured target.
  --fleet-public-key <string> — Path to the fleet server plan-signing public key PEM; agents refuse unverifiable plans without it (fail closed).
  --ssh-bootstrap-target <string> — Remote-SSH RunnerTarget id to connect and bootstrap as a fleet agent; empty disables SSH bootstrap.
  --secret-root <string> — Server-only root used to resolve file SecretRefs for remote-SSH bootstrap; never sent to the browser or remote host.
  --ssh-connect-timeout-ms <number> — SSH connection/bootstrap timeout in milliseconds.

Options (orchestrator):
  --kernel <string> — Kernel endpoint the orchestrator advances projects over. (default: http://127.0.0.1:7412)
  --db <string> — SQLite action-store path; empty = ephemeral temp database.
  --poll-ms <number> — Poll interval between advancement rounds (bin still requires > 0).
  --once <boolean> — Run a single poll round and exit (tests/cron).
  --dry-run <boolean> — Compute planned actions only; no kernel writes, no persistence.
  --owner <string> — Leader-election owner id for orchestrator_leases (reconstruction-contracts.md §15); empty = orch-<hostname>-<pid>.
  --lease-seconds <number> — orchestrator_leases expiry in seconds (§15); the owning instance refreshes every poll round and releases on close.
  --token-file <string> — 0600 file with the kernel bearer token; every kernel request carries Authorization: Bearer <token>.

Options (kernel):
  --host <string> — Kernel HTTP listen host. (default: 127.0.0.1)
  --port <number> — Kernel HTTP listen port (0 = ephemeral, published via endpoint file).
  --db <string> — SQLite database path; empty = ephemeral temp database.
  --cas <string> — CAS root for immutable artifacts. (default: .research-cas)
  --secret-root <secret> [secret] — Server-only SecretRef root for target identities, model credentials and SSH connection material; never returned in plaintext. (env: DSH_SCHOLAR_SECRET_ROOT)
  --endpoint-file <string> — 0600 endpoint identity file written after bind (SIDE-01). (env: DSH_SCHOLAR_KERNEL_ENDPOINT_FILE)

Options (standalone):
  --host <string> — Standalone BFF listen host. (default: 127.0.0.1) (env: DSH_SCHOLAR_STANDALONE_HOST)
  --port <number> — Standalone BFF listen port. (env: DSH_SCHOLAR_STANDALONE_PORT)
  --kernel-port <number> — Shared Research Kernel port used by both DSH and the standalone BFF. (env: DSH_SCHOLAR_STANDALONE_KERNEL_PORT)
  --kernel-data-dir <string> — Authoritative shared Kernel data directory; empty = $DSH_HOME/research-kernel or ~/.dsh/research-kernel. (env: DSH_SCHOLAR_KERNEL_DATA)
  --data-dir <string> — Standalone BFF state directory (browser token/session/model only); empty = $DSH_HOME/research-ui-standalone or ~/.dsh-scholar-standalone/research-ui-standalone. (env: DSH_SCHOLAR_STANDALONE_DATA)
  --principal <string> — Operator principal identity (GOV-01 local resolver).
  --frame-ancestors <string> — Comma-separated exact HTTPS or loopback-HTTP DSH origins allowed by CSP frame-ancestors; no wildcard, path, query, fragment, or credentials. (default: http://127.0.0.1:3080,http://localhost:3080,http://[::1]:3080) (env: DSH_SCHOLAR_STANDALONE_FRAME_ANCESTORS)
  --no-token <boolean> — Tokenless mode — loopback bind only (127.0.0.0/8, ::1, localhost).
